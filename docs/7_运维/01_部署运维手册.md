## 4. Docker部署

### 4.1 docker-compose.yml

```yaml
version: '3.8'

services:
  # 数据库
  postgres:
    image: postgres:16
    container_name: ai-bid-postgres
    environment:
      POSTGRES_DB: ai_bid
      POSTGRES_USER: ai_bid_admin
      POSTGRES_PASSWORD: ${DB_PASSWORD}
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./init-scripts:/docker-entrypoint-initdb.d
    networks:
      - ai-bid-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ai_bid_admin -d ai_bid"]
      interval: 10s
      timeout: 5s
      retries: 5

  # Redis
  redis:
    image: redis:7
    container_name: ai-bid-redis
    command: redis-server --requirepass ${REDIS_PASSWORD}
    ports:
      - "6379:6379"
    volumes:
      - redis_data:/data
    networks:
      - ai-bid-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5

  # MinIO
  minio:
    image: minio/minio
    container_name: ai-bid-minio
    environment:
      MINIO_ROOT_USER: ${MINIO_USER}
      MINIO_ROOT_PASSWORD: ${MINIO_PASSWORD}
    ports:
      - "9000:9000"
      - "9001:9001"
    volumes:
      - minio_data:/data
    networks:
      - ai-bid-network
    restart: unless-stopped
    command: server /data --console-address ":9001"
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:9000/minio/health/live"]
      interval: 30s
      timeout: 20s
      retries: 3

  # Chroma向量库
  chroma:
    image: chromadb/chroma
    container_name: ai-bid-chroma
    ports:
      - "8000:8000"
    volumes:
      - chroma_data:/chroma/chroma_db
    networks:
      - ai-bid-network
    restart: unless-stopped

  # 应用服务
  gateway:
    build: ./ai-bid-gateway
    container_name: ai-bid-gateway
    ports:
      - "8080:8080"
    environment:
      - SPRING_PROFILES_ACTIVE=${PROFILE}
      - DB_PASSWORD=${DB_PASSWORD}
      - REDIS_PASSWORD=${REDIS_PASSWORD}
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - ai-bid-network
    restart: unless-stopped

volumes:
  postgres_data:
  redis_data:
  minio_data:
  chroma_data:

networks:
  ai-bid-network:
    driver: bridge
```

## 5. 配置管理

### 5.1 应用配置 application.yml

```yaml
spring:
  application:
    name: ai-bid-system
  profiles:
    active: ${PROFILE:dev}
  datasource:
    url: jdbc:postgresql://postgres:5432/ai_bid
    username: ai_bid_admin
    password: ${DB_PASSWORD}
    driver-class-name: org.postgresql.Driver
  jpa:
    hibernate:
      ddl-auto: validate
    show-sql: false
  redis:
    host: redis
    port: 6379
    password: ${REDIS_PASSWORD}

server:
  port: ${SERVER_PORT:8080}
  servlet:
    context-path: /

# JWT配置
jwt:
  secret: ${JWT_SECRET}
  expiration: 7200
  refresh-expiration: 604800

# MinIO配置
minio:
  endpoint: http://minio:9000
  access-key: ${MINIO_USER}
  secret-key: ${MINIO_PASSWORD}
  bucket:
    material: ai-bid-materials
    document: ai-bid-documents

# AI服务配置
ai:
  llm:
    default-provider: minimax
    providers:
      minimax:
        api-key: ${MINIMAX_API_KEY}
        base-url: https://api.minimax.chat
      deepseek:
        api-key: ${DEEPSEEK_API_KEY}
        base-url: https://api.deepseek.com
  
  embedding:
    default-model: m3e
```

### 5.2 Nginx配置

```nginx
upstream ai-bid-gateway {
    server gateway:8080;
}

server {
    listen 80;
    server_name _;
    
    # 前端静态文件
    location / {
        root /usr/share/nginx/html;
        index index.html;
        try_files $uri $uri/ /index.html;
    }
    
    # API代理
    location /api/ {
        proxy_pass http://ai-bid-gateway;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # 超时配置
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }
    
    # WebSocket支持
    location /ws/ {
        proxy_pass http://ai-bid-gateway;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }
    
    # 静态资源缓存
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2)$ {
        root /usr/share/nginx/html;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
    
    # 健康检查
    location /health {
        access_log off;
        return 200 "OK";
    }
}
```

## 9. 备份恢复

### 9.1 自动备份脚本

```bash
#!/bin/bash
# backup.sh

BACKUP_DIR="/backups"
DATE=$(date +%Y%m%d_%H%M%S)
DB_CONTAINER="ai-bid-postgres"

# 创建备份目录
mkdir -p ${BACKUP_DIR}

# 数据库备份
docker exec ${DB_CONTAINER} pg_dump -U ai_bid_admin ai_bid > ${BACKUP_DIR}/ai_bid_${DATE}.sql

# 压缩备份
gzip ${BACKUP_DIR}/ai_bid_${DATE}.sql

# 保留最近30天备份
find ${BACKUP_DIR} -name "ai_bid_*.sql.gz" -mtime +30 -delete

echo "Backup completed: ai_bid_${DATE}.sql.gz"
```

### 9.2 定时备份（crontab）

```bash
# 每天凌晨2点执行备份
0 2 * * * /opt/ai-bid/backup.sh >> /var/log/backup.log 2>&1

# 每周日凌晨3点执行全量备份
0 3 * * 0 /opt/ai-bid/backup.sh full >> /var/log/backup.log 2>&1
```

### 9.3 数据恢复

```bash
# 1. 停止应用
docker-compose stop gateway

# 2. 恢复数据库
docker exec -i ai-bid-postgres psql -U ai_bid_admin -d ai_bid < ai_bid_20260525_020000.sql

# 3. 重启应用
docker-compose restart gateway
```

### 9.4 文件备份

```bash
# MinIO文件备份
docker run --rm -v ai-bid_minio_data:/data -v ${BACKUP_DIR}:/backup alpine tar czf /backup/minio_${DATE}.tar.gz -C /data .

# 恢复MinIO文件
docker run --rm -v ai-bid_minio_data:/data -v ${BACKUP_DIR}:/backup alpine tar xzf /backup/minio_${DATE}.tar.gz -C /data
```

## 10. 故障处理

### 10.1 常见故障

| 故障 | 症状 | 排查命令 | 解决方案 |
|------|------|----------|----------|
| 数据库连接失败 | 应用无法启动 | `docker logs gateway` | 检查DB_PASSWORD和网络 |
| Redis连接失败 | 会话失效 | `docker logs gateway` | 检查REDIS_PASSWORD |
| 文件上传失败 | 上传报错 | `docker logs minio` | 检查MinIO配置和存储空间 |
| 向量检索慢 | 查询超时 | `docker logs knowledge` | 增加内存，优化索引 |
| LLM调用失败 | 生成失败 | `docker logs ai` | 检查API Key和余额 |

### 10.2 紧急故障响应

```bash
# 1. 查看所有服务状态
docker-compose ps

# 2. 查看所有日志
docker-compose logs --tail=50

# 3. 重启故障服务
docker-compose restart [service_name]

# 4. 如服务异常，重建服务
docker-compose up -d --force-recreate --build [service_name]

# 5. 数据库紧急恢复
docker exec -i ai-bid-postgres psql -U ai_bid_admin -d ai_bid < latest_backup.sql
```

### 10.3 性能问题排查

```bash
# 查看资源使用
docker stats

# 查看进程
docker exec -it ai-bid-postgres ps aux

# 查看连接数
docker exec -i ai-bid-postgres psql -U ai_bid_admin -d ai_bid -c "SELECT count(*), state FROM pg_stat_activity GROUP BY state"

# 慢查询日志
docker exec -i ai-bid-postgres psql -U ai_bid_admin -d ai_bid -c "SELECT * FROM pg_stat_statements ORDER BY mean_exec_time DESC LIMIT 10"
```

## 11. 安全加固

### 11.1 服务器安全

```bash
# 1. 更新系统
apt update && apt upgrade -y

# 2. 配置防火墙
ufw default deny incoming
ufw default allow outgoing
ufw allow ssh
ufw allow http
ufw allow https
ufw enable

# 3. 配置SSH密钥登录（禁用密码）
vim /etc/ssh/sshd_config
# 设置 PasswordAuthentication no
# 设置 PubkeyAuthentication yes

# 4. 安装fail2ban防暴力破解
apt install fail2ban -y
systemctl enable fail2ban
systemctl start fail2ban
```

### 11.2 SSL证书配置

```bash
# 使用Let's Encrypt免费证书
apt install certbot python3-certbot-nginx -y
certbot --nginx -d yourdomain.com

# 自动续期
crontab -e
# 0 0 * * * certbot renew --quiet
```

### 11.3 容器安全

```yaml
# docker-compose.yml中添加安全配置
services:
  gateway:
    read_only: true
    security_opt:
      - no-new-privileges:true
```
